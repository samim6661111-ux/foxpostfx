cmake_minimum_required(VERSION 3.18)
project(foxpostfx CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fPIC")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fPIC")
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
FetchContent_Declare(dobby GIT_REPOSITORY https://github.com/jmpews/Dobby.git GIT_TAG 12e02a389ca7e9c1baf1e1eef082694144f57316)
FetchContent_MakeAvailable(dobby)
add_library(foxpostfx SHARED postfx.cpp)
target_include_directories(foxpostfx PRIVATE ${CMAKE_SOURCE_DIR}/gen ${dobby_SOURCE_DIR}/include)
target_link_libraries(foxpostfx dobby dl)
if(EXISTS /system/lib64/liblog.so)
  target_link_libraries(foxpostfx /system/lib64/liblog.so)
else()
  target_link_libraries(foxpostfx log)
endif()
target_link_options(foxpostfx PRIVATE -Wl,-z,max-page-size=16384)
